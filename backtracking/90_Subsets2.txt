90. Subsets 2
Approach: Backtracking

Solution:
class Solution(object):
    def subsetsWithDup(self, nums):
        nums.sort()
        result = []
        current = []
        def sub(index):
            result.append(current[:])
            for i in range(index, len(nums)):
                if i > index and nums[i] == nums[i-1]:
                    continue
                current.append(nums[i])
                sub(i+1)
                current.pop()
        sub(0)
        return result
