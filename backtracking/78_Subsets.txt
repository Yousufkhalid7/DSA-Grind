78. Subsets
Approach: Backtracking

Solution:
class Solution(object):
    def subsets(self, nums):
        result = []
        current = []
        def sub(index):
            result.append(current[:])
            for i in range(index, len(nums)):
                current.append(nums[i])
                sub(i + 1)
                current.pop()
        sub(0)
        return result
