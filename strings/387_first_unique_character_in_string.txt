387. First Unique Character in String
Approach: Hashmap

Solution:
class Solution(object):
    def firstUniqChar(self, s):
        umap = {}
        for character in s:
            if character in umap:
                umap[character] += 1
            else:
                umap[character] = 1
        for check in range(len(s)):
            if umap[s[check]] == 1:
                return check
        return -1
