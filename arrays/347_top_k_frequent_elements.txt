347. Top K Frequent Elements
Approach: HashMap

Solution:
class Solution(object):
    def topKFrequent(self, nums, k):
        hmap = {}
	result = []
	for num in nums:
		if num in hmap:
			hmap[num] += 1
		else:
			hmap[num] = 1
	sorted_hmap = sorted(hmap.items(), key = lambda x:x[1], reverse = True)
	for i in range(k):
		result.append(sorted_hmap[i][0])
	return result
