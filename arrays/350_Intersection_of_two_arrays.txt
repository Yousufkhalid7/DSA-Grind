350. Intersection of Two Arrays
Approach: Two Pointers

Solution:
class Solution(object):
    def intersect(self, nums1, nums2):
        hmap = {}
        result = []
        for num in nums1:
            if num in hmap:
                hmap[num] += 1
            else:
                hmap[num] = 1
        for num in nums2:
            if num in hmap and hmap[num] > 0:
                result.append(num)
                hmap[num] -= 1
        return result
