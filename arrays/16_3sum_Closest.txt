16. 3Sum Closest
Approach: Two Pointers + Check

Solution:
class Solution(object):
    def threeSumClosest(self, nums, target):
        nums.sort()
	current = nums[0] + nums[1] + nums[2]
	for i in range(len(nums)-2):
		l = i + 1
		r = len(nums) - 1
		while l < r:
			csum = nums[i] + nums[l] + nums[r]
			if abs(csum - target) < abs(current - target):
				current = csum
			if csum < target:
				l += 1
			elif csum > target:
				r -= 1
			else:
				return current
	return current
