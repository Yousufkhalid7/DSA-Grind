189. Rotate Array
Approach: D&C ( X )

Solution:
class Solution(object):
    def rotate(self, nums, k):
        k = k % len(nums)
        temp = nums[-k:]
        for i in reversed(range(0, len(nums)-k)):
            nums[i+k] = nums [i]
        for j in range(0, k):
            nums[j] = temp[j]          
        return nums

Approach: Reverse and Replace ( in-place )

class Solution(object):
    def rotate(self, nums, k):
        k = k % len(nums)
        n = len(nums)
        i = 0
        j = len(nums) - 1
        while i < j:
            nums[i], nums[j] = nums[j], nums[i]
            i += 1
            j -= 1
        i = 0 
        j = k - 1
        while i < j:
            nums[i], nums[j] = nums[j], nums[i]
            i += 1
            j -= 1
        i = k
        j = n - 1 
        while i < j:
            nums[i], nums[j] = nums[j], nums[i]
