128. Longest Consecutive Sequence
Approach: Sliding Window ( X O(nlogn) )

Solution:
class Solution(object):
    def longestConsecutive(self, nums):
        if not nums:
            return 0
        nums = list(set(nums))
        nums.sort()
        count = 1
        max_length = 1
        for i in range(1, len(nums)):
            if nums[i] - nums[i-1] == 1:
                count += 1
                max_length = max(max_length, count)
            else:
                count = 1
        return max_length

Approach: HashMap(O(logn))

Solution:
class Solution(object):
    def longestConsecutive(self, nums):
        nums_set = set(nums)
        max_count = 0
        for num in nums_set:
            if num - 1 not in nums_set:
                count = 1
                while num + count in nums_set:
                    count += 1
                max_count = max(count, max_count)
        return max_count
