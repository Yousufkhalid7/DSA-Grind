150. Reverse Polish Notation
Approach: Stack

Solution:
class Solution(object):
    def evalRPN(self, tokens):
        stack = []

        for ch in tokens:
            if ch == "+":
                stack.append(stack.pop() + stack.pop())

            elif ch == "-":
                a = stack.pop()
                b = stack.pop()
                stack.append(b - a)

            elif ch == "*":
                stack.append(stack.pop() * stack.pop())

            elif ch == "/":
                a = stack.pop()
                b = stack.pop()
                stack.append(int(b / a) if (b < 0) == (a < 0) else -(-b // a))

            else:
                stack.append(int(ch))

        return stack[0]
